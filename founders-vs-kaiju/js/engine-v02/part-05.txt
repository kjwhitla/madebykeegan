function queueTriggers(items,{front=false}={}){const jobs=(items||[]).filter(Boolean);if(!jobs.length)return;if(front)s.triggerQueue=[...jobs,...s.triggerQueue];else s.triggerQueue.push(...jobs);if(!s.processingTriggers&&!s.choice&&!s.pending)runTriggerQueue()}
function moveFortification(from,to,id){const src=area(from),i=src.forts.indexOf(id);if(i<0)return false;src.forts.splice(i,1);area(to).forts.push(id);log('fortify',card(id).name+' moves from '+src.label+' to '+area(to).label+'.');queueTriggers([{type:'enforce_fort_legality',loc:to}],{front:true});return true}
function runTriggerQueue(){if(s.processingTriggers||s.choice||s.pending||s.winner)return;s.processingTriggers=true;while(s.triggerQueue.length&&!s.choice&&!s.pending&&!s.winner){const job=s.triggerQueue.shift();resolveTrigger(job)}s.processingTriggers=false}
