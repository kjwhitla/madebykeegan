function gate(loc,mods={}){let g=baseGate(loc),a=area(loc);if((loc==='district1'||loc==='district2')&&a.state==='breached'){g.force=Math.max(1,g.force-1);g.adapt=Math.max(1,g.adapt-1)}if(a.state==='ruined')g={force:0,adapt:0};if(loc==='breach'&&s.waves.active.some(w=>w.location==='breach')){g.force=Math.max(1,g.force-1);g.adapt=Math.max(1,g.adapt-1)}if(loc==='cityCore'&&a.coreBreached){g.force=Math.max(1,g.force-1);g.adapt=Math.max(1,g.adapt-1)}for(const id of a.forts){if(id==='W01')g.force++;if(id==='W02'){g.force=Math.max(1,g.force-1);g.adapt++}if(id==='W06'){g.adapt=Math.max(1,g.adapt-1);g.force++}}if(s.kaiju.evolutions.head==='K13')g.force=Math.max(1,g.force-1);if(mods.reduceChosen&&mods.route)g[mods.route]=Math.max(0,g[mods.route]-mods.reduceChosen);if(mods.ignoreOne&&mods.route)g[mods.route]=Math.max(0,g[mods.route]-mods.ignoreOne);return g}

function summary(){if(!s.pending||!s.pending.route)return null;const p=s.pending,r=p.route,g=gate(p.next,{reduceChosen:p.reduceChosen,ignoreOne:p.ignoreOne,route:r});let got=r==='force'?p.bonusForce:p.bonusAdapt;for(const id of p.cards){const c=card(id);got+=id==='K09'?1:(c.gate[r]||0)}const miss=Math.max(0,g[r]-got),car=s.kaiju.evolutions.body==='K17'&&p.carapace&&miss>0?1:0;return{required:g[r],symbols:got,missing:miss,healthCost:miss-car}}

function confirmAdvance(){const p=s.pending,x=summary();if(!p||!x)return;p.cards.forEach(id=>discard('kaiju',id));if(p.carapace)discard('kaiju',p.carapace);s.metrics.routes[p.route]++;s.metrics.routes.overrun+=x.healthCost;if(x.healthCost)damage(x.healthCost,'Overrun');if(s.winner){s.pending=null;emit();return}const from=s.kaiju.location;s.kaiju.location=p.next;if(s.metrics.healthByEntry[p.next]==null)s.metrics.healthByEntry[p.next]=s.kaiju.health;log('advance',`Colossus ${from} → ${p.next} via ${p.route.toUpperCase()}.`,{route:p.route,from,to:p.next,overrunHealth:x.healthCost,cardsSpent:[...p.cards]});s.advanceContext={next:p.next,source:p.source,postDamage:p.postDamage,postDraw:p.postDraw,postResolved:false};s.pending=null;pushContinuation('finish_advance');resolveEntry(p.next,p.route,x.healthCost>0);resume();emit()}
function ruinDistrict(loc){const a=area(loc);if(!['district1','district2'].includes(loc)||a.state==='ruined')return;a.state='ruined';log('district',a.label+' becomes Ruined.');const hadW12=a.forts.includes('W12');const companion=a.forts.filter(id=>id!=='W12');const jobs=[];if(hadW12){removeFort(loc,'W12');s.washington.location=inward(s.washington.location);log('fallback','Orderly Retreat moves Washington inward.');if(companion.length)jobs.push({type:'w12_move_companion',loc,dest:inward(loc),ids:[...companion]})}jobs.push({type:'ruin_discard_remaining',loc});queueTriggers(jobs,{front:true})}
function breachDistrict(loc,{byWave=false,source=null}={}){const a=area(loc);if(!['district1','district2'].includes(loc)||a.state!=='fortified')return false;a.state='breached';log(byWave?'wave':'district',a.label+' becomes Breached.');if(byWave)s.metrics.wavesBreached++;const d=district(a.districtId),jobs=[];if(a.forts.length>1)jobs.push({type:'breach_capacity',loc,byWave});if(byWave&&d?.id==='D03')jobs.push({type:'d03_extra_loss',loc});if(byWave&&a.forts.includes('W17'))jobs.push({type:'w17_wave_breach',loc});if(d?.id==='D05'&&!a.marketWardTriggered){a.marketWardTriggered=true;jobs.push({type:'market_ward',loc})}if(source==='K15')jobs.push({type:'k15_breach_draw'});queueTriggers(jobs,{front:true});return jobs.length>0}
function resolveEntry(loc,route,overran){const a=area(loc),forts=[...a.forts],jobs=[];if(forts.includes('W09'))jobs.push({type:'entry_w09',loc});if(forts.includes('W16'))jobs.push({type:'entry_w16',loc});if(['district1','district2'].includes(loc)){const d=district(a.districtId);if(d.id==='D01'&&route==='force')jobs.push({type:'entry_damage',source:'Harbor Batteries',amount:1});if(d.id==='D02'&&route==='adapt')jobs.push({type:'d02_extra_discard'});if(d.id==='D06'&&route==='adapt')jobs.push({type:'d06_move_in',loc})}if(forts.includes('W01'))jobs.push({type:'entry_discard_fort',loc,id:'W01'});if(forts.includes('W02')&&route==='force')jobs.push({type:'entry_fort_damage',loc,id:'W02',source:'Earthwork Redoubt'});if(forts.includes('W03')&&route==='force')jobs.push({type:'entry_fort_damage',loc,id:'W03',source:'Coastal Battery'});if(forts.includes('W04')&&overran)jobs.push({type:'entry_fort_damage',loc,id:'W04',source:'Powder Magazine'});if(forts.includes('W07')&&route==='adapt')jobs.push({type:'w07_entry',loc});if(forts.includes('W11'))jobs.push({type:'w11_entry',loc});if(forts.includes('W13'))jobs.push({type:'w13_entry',loc});if(forts.includes('W05'))jobs.push({type:'w05_entry',loc});if(['district1','district2'].includes(loc))jobs.push({type:'body_deteriorate',loc});jobs.push({type:'advance_waves'});queueTriggers(jobs)}
